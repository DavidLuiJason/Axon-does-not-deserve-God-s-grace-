import { validateCharter, defaultCharter, CHARTER_LIMITS } from './charter.ts';
import type { Charter, Sha256, CharterError } from './charter.ts';
import { verifyCharter } from './change.ts';

export const CHARTER_UNTRUSTED_NOTICE = 'Your Charter could not be checked, so I ask before every action.';
const VIEW_MESSAGE = 'Charter view input is invalid.';
export type StoredCharterRead = { readonly kind: 'absent' } | { readonly kind: 'unreadable' } | { readonly kind: 'text'; readonly text: string };
export type UntrustedReason = CharterError | 'unreadable' | 'not-json' | 'unsigned' | 'signature-mismatch';
export type CharterLoad =
  | { readonly status: 'built-in'; readonly reason: 'no-stored-charter'; readonly charter: Charter }
  | { readonly status: 'signed'; readonly reason: 'trusted'; readonly charter: Charter }
  | { readonly status: 'untrusted'; readonly reason: UntrustedReason; readonly charter: null };
export type CharterLink = { readonly tool: string; readonly actions: readonly string[] };
export type CharterViewArgs = { readonly load: CharterLoad; readonly toolIds: readonly string[]; readonly links: readonly CharterLink[] };
export type CharterFeedView = { readonly never: readonly string[]; readonly ask: readonly string[]; readonly status: 'built-in' | 'signed' | 'charter-untrusted'; readonly notice: string };
type Fail<E extends string> = { readonly ok: false; readonly error: E; readonly message: string };
export type ViewResult = { readonly ok: true; readonly value: CharterFeedView } | Fail<'bad-input' | 'unknown-key' | 'bad-load' | 'bad-tool-ids' | 'bad-links' | 'unknown-link-tool'>;
export type SerializeResult = { readonly ok: true; readonly text: string } | Fail<CharterError>;

const isObj = (v: unknown): v is Record<PropertyKey, unknown> => v !== null && typeof v === 'object' && !Array.isArray(v);
const hasOwn = (o: object, k: PropertyKey): boolean => Object.prototype.hasOwnProperty.call(o, k);
const exact = (v: Record<PropertyKey, unknown>, keys: readonly string[]): boolean => { const own = Reflect.ownKeys(v); return own.length === keys.length && keys.every((k) => own.includes(k)); };
const strings = (v: unknown, distinct: boolean): v is string[] => Array.isArray(v) && v.every((s) => typeof s === 'string' && s.length > 0) && (!distinct || new Set(v).size === v.length);
const untrusted = (reason: UntrustedReason): CharterLoad => Object.freeze({ status: 'untrusted', reason, charter: null });

export function loadCharterText(stored: StoredCharterRead, sha256: Sha256): CharterLoad {
  try {
    if (typeof sha256 !== 'function' || !isObj(stored)) return untrusted('unreadable');
    if (stored.kind === 'absent' && exact(stored, ['kind'])) return Object.freeze({ status: 'built-in', reason: 'no-stored-charter', charter: defaultCharter() });
    if (stored.kind === 'text' && exact(stored, ['kind', 'text']) && typeof stored.text === 'string') {
      if (stored.text.length > CHARTER_LIMITS.inputCharacters) return untrusted('input-too-large');
      let parsed: unknown;
      try { parsed = JSON.parse(stored.text); } catch { return untrusted('not-json'); }
      const checked = validateCharter(parsed);
      if (!checked.ok) return untrusted(checked.error);
      const proof = verifyCharter(checked.value, sha256);
      if (!proof.ok) return untrusted(proof.reason as UntrustedReason);
      return Object.freeze({ status: 'signed', reason: 'trusted', charter: checked.value });
    }
    return untrusted('unreadable');
  } catch { return untrusted('unreadable'); }
}

export function serializeCharter(charter: unknown): SerializeResult {
  const checked = validateCharter(charter);
  if (!checked.ok) return Object.freeze({ ok: false, error: checked.error, message: checked.message });
  const text = JSON.stringify(checked.value, (_k, v) => (isObj(v) ? Object.fromEntries(Object.keys(v).sort().map((k) => [k, v[k]])) : v));
  return Object.freeze({ ok: true, text });
}

const RANK = { allowed: 0, ask: 1, never: 2 } as const;
type Lvl = keyof typeof RANK;

export function charterToFeedView(args: CharterViewArgs): ViewResult {
  const fail = (error: Exclude<ViewResult, { ok: true }>['error']): ViewResult => Object.freeze({ ok: false, error, message: VIEW_MESSAGE });
  if (!isObj(args)) return fail('bad-input');
  if (!exact(args, ['load', 'toolIds', 'links'])) return fail('unknown-key');
  const load = args.load as unknown;
  const loadOk = isObj(load) && ((load.status === 'built-in' && exact(load, ['status', 'reason', 'charter']) && load.reason === 'no-stored-charter') ||
    (load.status === 'signed' && exact(load, ['status', 'reason', 'charter']) && load.reason === 'trusted') ||
    (load.status === 'untrusted' && exact(load, ['status', 'reason', 'charter']) && typeof load.reason === 'string' && load.reason.length > 0 && load.charter === null));
  if (!loadOk) return fail('bad-load');
  if (!strings(args.toolIds, true)) return fail('bad-tool-ids');
  const links = args.links as unknown;
  if (!Array.isArray(links) || !links.every((l) => isObj(l) && exact(l, ['tool', 'actions']) && typeof l.tool === 'string' && l.tool.length > 0 && strings(l.actions, true)) ||
    new Set(links.map((l) => (l as CharterLink).tool)).size !== links.length) return fail('bad-links');
  if (!links.every((l) => args.toolIds.includes((l as CharterLink).tool))) return fail('unknown-link-tool');

  const l = load as CharterLoad;
  const closed = (): ViewResult => Object.freeze({ ok: true, value: Object.freeze({ never: Object.freeze([]), ask: Object.freeze([...args.toolIds]), status: 'charter-untrusted', notice: CHARTER_UNTRUSTED_NOTICE }) as CharterFeedView });
  if (l.status === 'untrusted') return closed();
  const checked = validateCharter(l.charter);
  if (!checked.ok) return closed();
  const lanes = checked.value.lanes;
  const laneIds = Object.keys(lanes);
  const levelsOf = (actionId: string): Lvl[] => laneIds.filter((id) => hasOwn(lanes[id].actions, actionId)).map((id) => lanes[id].actions[actionId].level as Lvl);
  const never: string[] = []; const ask: string[] = [];
  for (const tool of args.toolIds) {
    let worst = 0;
    for (const lv of levelsOf(`tools.${tool}`)) worst = Math.max(worst, RANK[lv]);
    const link = (links as CharterLink[]).find((x) => x.tool === tool);
    for (const action of link ? link.actions : []) {
      const found = levelsOf(action);
      if (found.length === 0) worst = Math.max(worst, RANK.ask);
      for (const lv of found) worst = Math.max(worst, RANK[lv]);
    }
    if (worst === RANK.never) never.push(tool); else if (worst === RANK.ask) ask.push(tool);
  }
  const status = l.status === 'signed' ? 'signed' : 'built-in';
  return Object.freeze({ ok: true, value: Object.freeze({ never: Object.freeze(never), ask: Object.freeze(ask), status, notice: '' }) as CharterFeedView });
}
