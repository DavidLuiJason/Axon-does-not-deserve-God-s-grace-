## 2. THE ONE ENTRY POINT
import { createNativeScreenSystem } from '../axon-system/screen-system-native';  const sys = createNativeScreenSystem({ policy, onEvent });  (create it ONCE, not per render).
- sys.status() -> { readerOn, canAct, appCount, appsCode, stopRequested, killSwitchAvailable, running }
- sys.screen.openSettings() opens Android Accessibility settings (the owner switches "AXON screen reader" on).
- Apps: sys.controller.listApps(query?) -> { ok, apps:[{ packageName, label, system, self, category, versionName }] }; sys.apps.getAppIcon(pkg, size) -> { ok, png (base64) }; sys.controller.open(nameOrAlias) -> status done | needs-approval | ambiguous (candidates) | not-found | failed | opened-not-seen; sys.apps.openAppSettings(pkg); sys.profiles.setAlias(name, pkg) / removeAlias / setNote / toJSON (persist with createProfilesPersistence in axon-autoclicker/routine-backend.mjs); sensitive badge: sensitiveApps(catalog) from axon-apps/app-catalog.mjs.
- Acting: sys.controller.see({pictures}), tap(label), type(label,text), toggle, scroll(direction), back(), home(), run(steps), resume(bool), approve(bool), stop(), log() -> [{ at, text, outcome }].
- Auto-clicker: sys.store.list/get/save/approve/remove/setSchedule; sys.runner.preview(routine) -> { ok, errors | scope, repeat, everyMs, limits, lines }; sys.runner.run(routine) -> { status: finished | stopped | failed | needs-approval | not-approved | invalid | refused | busy, reason, loops, actions, ms, log | question | errors }; sys.runner.answer(bool); sys.runner.stop(); sys.recorder.start({ blockedPackages }) / stop({ name, scope }) -> { ok, routine (DRAFT), notes, truncated }. Live progress comes through onEvent({ type: started | step | needs-approval | finished | scheduled-start, ... }).
- Self-test: sys.selfTest({ approveTyping }) -> { passed, lines:[{ step, status PASS|FAIL|SKIP, text }], summary }.
